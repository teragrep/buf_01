167 @Override
168 public boolean hasNext() {
169 final boolean rv;
170 if (limit == -1) {
171 // limit not set, ignore
172 rv = currentOffset < origin.leasedObject().byteSize();
173 }
174 else {
175 rv = currentOffset < Math.min(limit, origin.leasedObject().byteSize());
176 }

204 public void position(final long newPosition) {
205 final long segmentByteSize = leasedObject().byteSize();
206 if (newPosition < 0 || newPosition > segmentByteSize) {
207 throw new IndexOutOfBoundsException(
208 "New position was out of bounds; expected value between 0 and " + segmentByteSize + "; was "

212
213 final long currentLimit = limit;

215 throw new IndexOutOfBoundsException(
